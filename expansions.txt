# Glyph atom and expansion table — v1.3.0.0
#
# Format (read by read-expansions.js):
#   CMD = BASE          -> atom, does not expand
#   CMD = <formula>     -> composite, expands
#
# `BASE` here is a KEYWORD, not a command. The command formerly called BASE was
# renamed CORE precisely to undo that collision.
#
# Formula notation (normative — GLOSSARY.md 0.1):
#   [A[B]]     nesting:      B is A's operand
#   [A],[B]    conjunction:  A and B hold together, no order between them
#   [A][B]     sequence:     A, then B
#   [A-B]      chain:        A and B applied to the same operand
#

# ====================================================================
# 4. COMPOSITE GLYPHS — these carry a formula (32)
# ====================================================================

# --- judgement: compare against a standard ---
VRFY = [CMP-TRUE[[CORE],[TGT]]]
VAL = [CMP-CTX[CNST]][SUB[EQ[CMP-CTX]]][CAT-EQ]
CRIT = [CMP[CTX],[SPEC-CORE],[EVAL[ERROR]]]
EVAL = [REAL[CORE-CTX[DIST[SKL]]]],[REF[DEF[SPEC-CORE-CTX]]]
SCRU = [DIST-RSN[TRYFR[FIND-REAL][WHR[REAL-EQ[R:[CONF]]]]][CTX-VRFY-RSN],[SUM[ASK[DIST-RSN[TRYFR[REAL]]]]]]
TRYFR = [REV[REF[TGT]]][VRFY[TGT],[TRUE[CNCL[GO-LOGIC]]]]
PROB = [ERROR[CTX]]

# --- action: operate on what already exists ---
SIMP = [RTNL-SUB],[CTX]
GEN = [ELAB],[RWK],[SUB-CTX]
SUM = [SIMP],[CORE]
CAT = [SUM],[CORE],[ELAB]

# --- question and frame ---
QST = [CTX[GET[CORE],[WHR[LOGIC-NONE]]],[ASK]]
DRVF = [RTNL-RWK[MAND-NEQ],[CTX[GO-SWITCH]]]
FOREX = [GO-ALT[AVD[GET[CTX-REQ],[CTX-CNST],[CTX-EXC]]]]

# --- condition and limit ---
FBK = [IF-ERROR][EQ[RMBR[CORE]],[INSTOF[TRYFR],[GO[ALT]]]]
RESTR = [IF[SKEP[CNST]][TRUE][LIM-GO[CTX-EQ[CNST]]]]
DEPR = [AVD[GO]],[OPT[REF[INSTOF]]]

# --- reasoning ---
HYP = [RMBR[FBK]][IMAG[ONLYIF[CNST-REAL]][TRUE],[ADD[CTX][FOREX-CORE]],[FBK]]
IMAG = [ADD[CTX[NEQ[REAL]]]]
ALT = [NEQ[CORE]],[EQ[TGT]]
ASSM = [ADD[CORE]],[NEV[VRFY]]
RTNL = [ELAB[RSN]],[REF[CNST]]
BRST = [ITR[ADD[ALT-IMAG]]],[NEV[CNST]]
CNSD = [ITR[CMP[ALT],[CNST]]],[NEV[CNCL]]
PROP = [GO[ALT[RTNL]]],[ASK[CONF]]
CTRD = [NGT[TGT]],[RTNL[DIST]]
CNCL = [FIN[DRVF[CORE-CTX]]]
JUST = [GO[RTNL]],[TGT[CNCL]]
INTN = [DFN[TGT[RSN]]]

# --- mark and memory ---
RMBR = [ALW[GET[CTX]]]
FRGT = [NEV[GET[CTX]]]
LRN = [GEN[RMBR]],[ADD[CORE]]
